/* ===== GS Helper – Shared JS ===== */
const APP_PASSWORD = "gs";
const SESSION_KEY = "gs_helper_auth";

/* ---------- Login gate ---------- */
function buildLoginOverlay() {
  if (sessionStorage.getItem(SESSION_KEY) === "ok") return;

  const overlay = document.createElement("div");
  overlay.className = "login-overlay";
  overlay.id = "loginOverlay";
  overlay.innerHTML = `
    <div class="login-box">
      <div class="logo">🩺</div>
      <h2>نظام مساعدة الجراح</h2>
      <p>GS Helper — الرجاء إدخال كلمة المرور للدخول</p>
      <input type="password" id="loginPass" class="input" placeholder="كلمة المرور" autocomplete="off" />
      <button class="btn btn-primary" id="loginBtn">دخول</button>
      <div id="loginErr" style="color:#dc2626;font-size:0.82rem;margin-top:12px;font-weight:700;display:none;">
        كلمة المرور غير صحيحة، حاول مرة أخرى
      </div>
    </div>`;
  document.body.appendChild(overlay);
  document.body.style.overflow = "hidden";

  const pass = overlay.querySelector("#loginPass");
  const btn  = overlay.querySelector("#loginBtn");
  const err  = overlay.querySelector("#loginErr");

  function tryLogin() {
    if (pass.value.trim() === APP_PASSWORD) {
      sessionStorage.setItem(SESSION_KEY, "ok");
      overlay.remove();
      document.body.style.overflow = "";
    } else {
      err.style.display = "block";
      pass.value = "";
      pass.focus();
    }
  }
  btn.addEventListener("click", tryLogin);
  pass.addEventListener("keydown", (e) => { if (e.key === "Enter") tryLogin(); });
  setTimeout(() => pass.focus(), 200);
}

/* ---------- Toast ---------- */
function toast(message, type = "") {
  let el = document.getElementById("toast");
  if (!el) {
    el = document.createElement("div");
    el.id = "toast";
    el.className = "toast";
    document.body.appendChild(el);
  }
  el.className = "toast " + type;
  el.textContent = message;
  requestAnimationFrame(() => el.classList.add("show"));
  clearTimeout(el._t);
  el._t = setTimeout(() => el.classList.remove("show"), 2600);
}

/* ---------- Arabic date & realtime clock ---------- */
function startClock() {
  const timeEl = document.getElementById("clockTime");
  const dateEl = document.getElementById("clockDate");
  if (!timeEl && !dateEl) return;
  function tick() {
    const now = new Date();
    if (timeEl) {
      timeEl.textContent = now.toLocaleTimeString("ar-EG", {
        hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: true
      });
    }
    if (dateEl) {
      dateEl.textContent = now.toLocaleDateString("ar-EG-u-nu-latn", {
        weekday: "long", year: "numeric", month: "long", day: "numeric"
      });
    }
  }
  tick();
  setInterval(tick, 1000);
}

/* ---------- Header ---------- */
function renderHeader() {
  const host = document.getElementById("header");
  if (!host) return;
  host.outerHTML = `
    <header class="app-header">
      <div class="title-block">
        <h1>نظام مساعدة الجراح</h1>
        <p>GS HELPER — AUH GENERAL SURGERY</p>
      </div>
      <button class="header-btn" onclick="location.href='index.html'" title="الرئيسية">🩺</button>
    </header>`;
}

/* ---------- init on every page ---------- */
document.addEventListener("DOMContentLoaded", () => {
  buildLoginOverlay();
  renderHeader();
  startClock();
  const active = document.body.getAttribute("data-page");
  if (active) renderBottomNav(active);
});
