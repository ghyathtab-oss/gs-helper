let TERMS = [];

function renderTerms(list) {
  const host = document.getElementById("termsList");
  if (!list.length) { host.innerHTML = `<div class="empty">لا توجد مصطلحات مطابقة</div>`; return; }
  host.innerHTML = list.map(t => `
    <div class="term-card">
      <div class="en">${t.en || ""}</div>
      <div class="ar">${t.ar || ""}</div>
      ${t.def ? `<div class="def">${t.def}</div>` : ""}
      ${t.ex ? `<div class="ex">💡 ${t.ex}</div>` : ""}
    </div>`).join("");
}

async function loadTerms() {
  try {
    const r = await fetch("assets/data/terms.json", { cache: "no-store" });
    TERMS = await r.json();
  } catch { TERMS = []; }
  renderTerms(TERMS);
}

document.addEventListener("DOMContentLoaded", () => {
  loadTerms();
  document.getElementById("searchTerms").addEventListener("input", (e) => {
    const q = e.target.value.trim().toLowerCase();
    if (!q) return renderTerms(TERMS);
    renderTerms(TERMS.filter(t => [t.en, t.ar, t.def, t.ex].some(v => String(v||"").toLowerCase().includes(q))));
  });
});
