let RESIDENTS = [];

          <stop offset='0' stop-color='%23ec4899'/><stop offset='1' stop-color='%23f472b6'/>
        </linearGradient></defs>
        <rect width='100' height='100' rx='22' fill='url(%23g)'/>
        <circle cx='50' cy='40' r='16' fill='%23fff'/>
        <path d='M34 38c0-12 8-20 16-20s16 8 16 20c2-1 4 1 4 4s-3 5-5 4c-1 6-7 12-15 12s-14-6-15-12c-2 1-5-1-5-4s2-5 4-4z' fill='%23fff'/>

function cleanPhone(p) { return String(p || "").replace(/[^\d+]/g, ""); }
function waNumber(p) {
  let n = cleanPhone(p).replace(/^\+/, "");
  if (n.startsWith("00")) n = n.slice(2);
  if (n.startsWith("0")) n = "963" + n.slice(1);
  return n;
}

function renderResidents(list) {
  const host = document.getElementById("residentsList");
  if (!list.length) { host.innerHTML = `<div class="empty">لا توجد نتائج مطابقة</div>`; return; }
  host.innerHTML = list.map(r => {
    const female = String(r.gender || "").trim().toUpperCase().startsWith("F");
    const phone = cleanPhone(r.phone);
    return `
      <div class="resident-card">
        <img class="avatar ${female ? "female" : "male"}" src="${avatarFor(r.gender)}" alt="avatar" />
        <div class="r-info">
          <h3>${r.nameAr || "—"}</h3>
          <div class="en">${r.nameEn || ""}</div>
          <span class="r-badge ${female ? "female" : ""}">${r.year || "—"}</span>
          <div class="r-phone">${phone || "—"}</div>
        </div>
        <div class="r-actions">
          ${phone ? `<a class="icon-btn icon-call" href="tel:${phone}" title="اتصال">📞</a>` : ""}
          ${phone ? `<a class="icon-btn icon-wa" href="https://wa.me/${waNumber(phone)}" target="_blank" rel="noopener" title="واتساب">💬</a>` : ""}
        </div>
      </div>`;
  }).join("");
}

function applySearch() {
  const q = document.getElementById("searchResidents").value.trim().toLowerCase();
  if (!q) return renderResidents(RESIDENTS);
  renderResidents(RESIDENTS.filter(r =>
    [r.nameAr, r.nameEn, r.year, r.phone].some(v => String(v || "").toLowerCase().includes(q))
  ));
}

async function loadResidents() {
  try {
    const res = await fetch("residents.xlsx", { cache: "no-store" });
    if (!res.ok) throw new Error("not found");
    const buf = await res.arrayBuffer();
    const wb = XLSX.read(buf, { type: "array" });
    const sheet = wb.Sheets[wb.SheetNames[0]];
    const rows = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: "" });
    const data = rows.slice(1).filter(r => r && (r[0] || r[1]))
      .map(r => ({ nameAr: r[0], nameEn: r[1], year: r[2], phone: r[3], gender: r[4] }));
    RESIDENTS = data.length ? data : SAMPLE_RESIDENTS;
  } catch { RESIDENTS = SAMPLE_RESIDENTS; }
  renderResidents(RESIDENTS);
}

document.addEventListener("DOMContentLoaded", () => {
  loadResidents();
  document.getElementById("searchResidents").addEventListener("input", applySearch);
});
